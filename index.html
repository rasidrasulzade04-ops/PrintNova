// Функция расчета себестоимости по всем слотам function calculatePrintCost() { let totalMaterialCost = 0; // Проходим по всем 4 слотам for (let i = 1; i <= 4; i++) { // Получаем вес в граммах для текущего слота const weightInput = document.getElementById(`weight${i}`); // Получаем цену за 1 кг (или за грамм, в зависимости от вашей разметки) для текущего слота const priceInput = document.getElementById(`price${i}`); if (weightInput && priceInput) { const weight = parseFloat(weightInput.value) || 0; // вес в граммах const pricePerKg = parseFloat(priceInput.value) || 0; // цена за кг // Вычисляем стоимость пластика для этого конкретного слота (цена за грамм * вес) const pricePerGram = pricePerKg / 1000; const slotCost = weight * pricePerGram; // Суммируем себестоимость этого слота в общую копилку totalMaterialCost += slotCost; } } // Если у вас есть фиксированная наценка, электричество или маржа, их можно добавить здесь let finalCost = totalMaterialCost; // Выводим результат на экран (замените 'total-price' на ID вашего элемента на сайте) const resultElement = document.getElementById('total-price'); if (resultElement) { resultElement.innerText = finalCost.toFixed(2) + " AZN"; } return finalCost; } // Привязываем автообновление при изменении полей в инпутах document.querySelectorAll('.slot-input').forEach(input => { input.addEventListener('input', calculatePrintCost); });
